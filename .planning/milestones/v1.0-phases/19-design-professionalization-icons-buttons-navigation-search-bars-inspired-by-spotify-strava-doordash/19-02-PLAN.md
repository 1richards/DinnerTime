       ```typescript
       import React from 'react';
       import { Pressable, Text, ActivityIndicator, type PressableProps } from 'react-native';
       import { SymbolView, type SymbolViewProps } from 'expo-symbols';
       import { variantStyles, type ButtonVariant } from './buttonStyles';
       import { iconPropsForText } from '../../design/icons';

       interface ButtonProps extends Omit&lt;PressableProps, 'children'&gt; {
         title?: string;
         variant?: ButtonVariant;
         loading?: boolean;
         icon?: SymbolViewProps['name'];  // required when variant='iconOnly'
         className?: string;
       }

       export function Button({
         title,
         variant = 'primary',
         loading = false,
         disabled = false,
         icon,
         className = '',
         ...props
       }: ButtonProps) {
         const s = variantStyles[variant];
         const isDisabled = disabled || loading;

         return (
           &lt;Pressable
             className={`${s.container} ${isDisabled ? 'opacity-50' : ''} ${className}`}
             disabled={isDisabled}
             accessibilityRole="button"
             {...props}
           &gt;
             {loading ? (
               &lt;ActivityIndicator color={s.spinnerColor} size="small" /&gt;
             ) : variant === 'iconOnly' &amp;&amp; icon ? (
               &lt;SymbolView name={icon} {...iconPropsForText('body')} tintColor={s.spinnerColor} /&gt;
             ) : (
               &lt;Text className={s.text}&gt;{title}&lt;/Text&gt;
             )}
           &lt;/Pressable&gt;
         );
       }

       export type { ButtonVariant };
       ```

    1. Create `apps/mobile/src/components/ui/chipStyles.ts`:

       ```typescript
       export type ChipKind = 'filter' | 'display';
       export type ChipTone = 'default' | 'success' | 'warning' | 'destructive';

       interface ResolveArgs {
         kind: ChipKind;
         selected?: boolean;     // filter only
         tone?: ChipTone;        // display only
       }

       const BASE_CONTAINER = 'h-8 px-3 rounded-pill flex-row items-center';
       const BASE_LABEL = 'text-caption';

       export function resolveChipClasses(args: ResolveArgs): { container: string; label: string } {
         if (args.kind === 'filter') {
           if (args.selected) {
             return {
               container: `${BASE_CONTAINER} bg-brand`,
               label: `${BASE_LABEL} text-white font-semibold`,
             };
           }
           return {
             container: `${BASE_CONTAINER} bg-surface border border-border`,
             label: `${BASE_LABEL} text-text-primary`,
           };
         }
         // display kind
         const tone = args.tone ?? 'default';
         switch (tone) {
           case 'success':
             return { container: `${BASE_CONTAINER} bg-success/15`, label: `${BASE_LABEL} text-success font-semibold` };
           case 'warning':
             return { container: `${BASE_CONTAINER} bg-warning/15`, label: `${BASE_LABEL} text-warning font-semibold` };
           case 'destructive':
             return { container: `${BASE_CONTAINER} bg-destructive/15`, label: `${BASE_LABEL} text-destructive font-semibold` };
           default:
             return { container: `${BASE_CONTAINER} bg-surface-subtle`, label: `${BASE_LABEL} text-text-secondary` };
         }
       }
       ```

       ```typescript
       import React from 'react';
       import { Pressable, View, Text } from 'react-native';
       import { SymbolView, type SymbolViewProps } from 'expo-symbols';
       import { resolveChipClasses, type ChipKind, type ChipTone } from './chipStyles';
       import { iconPropsForText } from '../../design/icons';
       import { colors } from '../../design/tokens';

       interface ChipProps {
         label: string;
         kind: ChipKind;
         selected?: boolean;
         onPress?: () =&gt; void;
         tone?: ChipTone;
         leadingIcon?: SymbolViewProps['name'];
       }

       export function Chip({ label, kind, selected, onPress, tone, leadingIcon }: ChipProps) {
         const { container, label: labelCls } = resolveChipClasses({ kind, selected, tone });
         const isInteractive = kind === 'filter' &amp;&amp; typeof onPress === 'function';
         const Wrapper = isInteractive ? Pressable : View;

         return (
           &lt;Wrapper
             onPress={isInteractive ? onPress : undefined}
             className={container}
             accessibilityRole={isInteractive ? 'button' : undefined}
           &gt;
             {leadingIcon ? (
               &lt;SymbolView
                 name={leadingIcon}
                 {...iconPropsForText('caption')}
                 tintColor={kind === 'filter' &amp;&amp; selected ? '#FFFFFF' : colors.textSecondary}
                 style={{ marginRight: 4 }}
               /&gt;
             ) : null}
             &lt;Text className={labelCls}&gt;{label}&lt;/Text&gt;
           &lt;/Wrapper&gt;
         );
       }

       export type { ChipKind, ChipTone };
       ```
