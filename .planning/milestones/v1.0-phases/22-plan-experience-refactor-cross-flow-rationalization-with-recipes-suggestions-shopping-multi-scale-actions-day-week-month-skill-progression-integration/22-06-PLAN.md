    Implement:
    ```ts
    export const PANTRY_STAPLES: ReadonlySet<string> = new Set([
      'salt', 'pepper', 'water', 'oil', 'olive oil', 'vegetable oil',
      'butter', 'sugar', 'flour', 'garlic powder', 'onion powder',
    ]);

    const norm = (s: string) => s.trim().toLowerCase();

    export function computePantryReady(
      ingredients: MealPlanIngredient[],
      pantryItems: PantryItem[],
    ): boolean {
      const pantryNames = new Set(pantryItems.map((p) => norm(p.name)));
      const nonStaple = ingredients
        .map((i) => norm(i.name ?? ''))
        .filter((n) => n && !PANTRY_STAPLES.has(n));

      if (ingredients.length === 0) return false;
      if (nonStaple.length === 0) return true;   // everything is a staple

      let matched = 0;
      for (const name of nonStaple) {
        for (const p of pantryNames) {
          if (p.includes(name) || name.includes(p)) { matched++; break; }
        }
      }
      return matched / nonStaple.length >= 0.8;
    }
    ```

    ```ts
    mealPlans.post('/:id/entries/:day/skip', async (c) => {
      const supabase = c.get('supabase');
      const user = c.get('user');
      const planId = c.req.param('id');
      const day = Number.parseInt(c.req.param('day') ?? '', 10);
      if (!Number.isInteger(day) || day < 0 || day > 6) {
        return c.json({ error: 'day must be 0..6' }, 400);
      }
      let body: { reason?: string | null } = {};
      try { body = await c.req.json(); } catch {}
      // Verify plan ownership
      const { data: plan } = await supabase
        .from('meal_plans')
        .select('id')
        .eq('id', planId)
        .eq('profile_id', user.id)
        .maybeSingle();
      if (!plan) return c.json({ error: 'Not found' }, 404);

      const { data: updated, error } = await supabase
        .from('meal_plan_entries')
        .update({ status: 'skipped', skip_reason: body.reason ?? null })
        .eq('meal_plan_id', planId)
        .eq('day_of_week', day)
        .select()
        .maybeSingle();
      if (error) return c.json({ error: error.message }, 500);
      if (!updated) return c.json({ error: 'Entry not found' }, 404);
      return c.json({ data: updated });
    });
    ```

       ```tsx
       import React from 'react';
       import { View, Text, Pressable, StyleSheet } from 'react-native';
       import ReanimatedSwipeable from 'react-native-gesture-handler/ReanimatedSwipeable';
       import { SymbolIcon } from '../ui/SymbolIcon';
       import { colors } from '../../design/tokens';
       import { DayRow } from './DayRow';
       import { logPlanEvent, sanitizePayload } from '../../plan/telemetry';
       import type { MealPlanEntry } from '../../types/mealPlan';

       export interface SwipeableDayRowProps {
         entry: MealPlanEntry | null;
         dayLabel: string;
         isSwapping: boolean;
         isCooking: boolean;
         onSwap: () => void;
         onCook: () => void;
         onSkip: () => void;
         onPress: () => void;
       }

       function Action({
         icon, label, tint, onPress,
       }: { icon: string; label: string; tint: string; onPress: () => void }) {
         return (
           <Pressable onPress={onPress} style={[styles.action, { backgroundColor: tint }]} accessibilityLabel={label}>
             <SymbolIcon name={icon as any} size={20} tintColor="#FFFFFF" />
             <Text style={styles.actionLabel}>{label}</Text>
           </Pressable>
         );
       }

       export function SwipeableDayRow(props: SwipeableDayRowProps) {
         const { entry, onSwap, onCook, onSkip } = props;
         if (!entry) return <DayRow {...props} />;

         const fire = (variant: 'swap' | 'cook' | 'skip', fn: () => void) => () => {
           logPlanEvent({
             name: 'plan.swipe_action',
             session_id: `swipe-${entry.id}`,
             meal_plan_id: entry.meal_plan_id,
             meal_plan_entry_id: entry.id,
             payload: sanitizePayload({ variant, meal_plan_entry_id: entry.id }),
           });
           fn();
         };

         return (
           <ReanimatedSwipeable
             renderRightActions={() => (
               <View style={styles.actionGroup}>
                 <Action icon="arrow.left.arrow.right" label="Swap" tint={colors.brand} onPress={fire('swap', onSwap)} />
                 <Action icon="checkmark" label="Cooked" tint={colors.success} onPress={fire('cook', onCook)} />
                 <Action icon="xmark" label="Skip" tint={colors.warning} onPress={fire('skip', onSkip)} />
               </View>
             )}
           >
             <DayRow {...props} />
           </ReanimatedSwipeable>
         );
       }

       const styles = StyleSheet.create({
         actionGroup: { flexDirection: 'row' },
         action: {
           width: 72,
           alignItems: 'center',
           justifyContent: 'center',
           gap: 4,
         },
         actionLabel: { fontSize: 11, fontWeight: '700', color: '#FFFFFF' },
       });
       ```
