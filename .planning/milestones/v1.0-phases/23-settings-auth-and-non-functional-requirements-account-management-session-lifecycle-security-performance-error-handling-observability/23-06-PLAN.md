    ```typescript
    import * as Sentry from '@sentry/react-native';

    const PII_KEY_RE = /email|password|token|transcript|raw_query|prompt|display_name|name/i;

    function stripPII(obj: Record<string, unknown> | undefined, depth = 0): Record<string, unknown> {
      if (!obj || depth > 2) return obj ?? {};
      const out: Record<string, unknown> = {};
      for (const [k, v] of Object.entries(obj)) {
        if (PII_KEY_RE.test(k)) continue;
        out[k] = (v && typeof v === 'object' && !Array.isArray(v))
          ? stripPII(v as Record<string, unknown>, depth + 1)
          : v;
      }
      return out;
    }

    export function initSentry(dsn: string | undefined): void {
      if (!dsn) return;
      Sentry.init({
        dsn,
        tracesSampleRate: __DEV__ ? 0.1 : 0.2,
        beforeSend(event) {
          if (event.extra) event.extra = stripPII(event.extra);
          return event;
        },
      });
    }

    export function setSentryUser(userId: string | null): void {
      Sentry.setUser(userId ? { id: userId } : null);
    }

    export function captureBreadcrumb(category: string, message: string, data?: Record<string, unknown>) {
      Sentry.addBreadcrumb({ category, message, data: data ? stripPII(data) : undefined, level: 'info' });
    }

    export function captureException(err: unknown, options?: { tags?: Record<string, string> }) {
      Sentry.captureException(err, options);
    }
    ```

    function shortId(): string { return Math.random().toString(36).slice(2, 14); }

    export async function requestLoggingMiddleware(c: Context, next: Next) {
      const reqId = shortId();
      const t0 = Date.now();
      c.set('request_id', reqId);
      try { await next(); }
      finally {
