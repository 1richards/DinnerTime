    ```typescript
    export const STARTUP_COLD_MS = 2000;
    export const TAB_SWITCH_MS = 16;
    export const SCAN_FEEDBACK_MS = 500;
    export const SCAN_COMPLETE_MS = 6000;
    export const RECEIPT_COMPLETE_MS = 8000;
    export const IMAGE_MAX_MB = 5;

    export async function withBudget<T>(name: string, budgetMs: number, fn: () => Promise<T>): Promise<T> {
      const t0 = Date.now();
      try { return await fn(); }
      finally {
        const ms = Date.now() - t0;
        if (ms > budgetMs) {
          try {
            // eslint-disable-next-line @typescript-eslint/no-require-imports
            const { captureBreadcrumb } = require('./sentry');
            captureBreadcrumb('perf', `${name}:over_budget`, { ms, budget_ms: budgetMs });
          } catch { /* sentry not wired */ }
          if (__DEV__) {
            // eslint-disable-next-line no-console
            console.warn(`[perf] ${name} took ${ms}ms (budget ${budgetMs}ms)`);
          }
        }
      }
    }
    ```
