    ```typescript
    export const ALLOWED_DEEP_LINK_PATHS = [
      /^\/recipes(\/|$)/,
      /^\/scan(\/|$)/,
      /^\/auth\/reset-password(\/|$)/,
      /^\/plan(\/|$)/,
      /^\/settings(\/|$)/,
      /^\/$/,
    ] as const;

    export function isDeepLinkAllowed(url: string): boolean {
      try {
        // Support both dinnertime:// and https://dinnertime.app URLs
        let path: string;
        if (url.startsWith('dinnertime://')) {
          const rest = url.slice('dinnertime://'.length);
          path = '/' + rest.split(/[?#]/)[0];
        } else if (url.startsWith('http')) {
          const parsed = new URL(url);
          path = parsed.pathname;
        } else {
          return false;
        }
        return ALLOWED_DEEP_LINK_PATHS.some((re) => re.test(path));
      } catch { return false; }
    }
    ```
