      const plan = get().currentPlan;
      if (!plan) return;
      const snapshot = plan.entries;
      // optimistic
      set({
        currentPlan: { ...plan, entries: plan.entries.map(e =>
          e.day_of_week === day ? { ...e, status: 'cooked' } : e
        )},
        cookingDay: day,
      });
      try {
        const res = await authedFetch(`/meal-plans/${plan.id}/entries/${day}/cook`, { method: 'POST' });
        // replace with server truth
      } catch (err) {
        // rollback
        set({ currentPlan: { ...plan, entries: snapshot }, error: ... });
      } finally {
        set({ cookingDay: null });
      }
    }
