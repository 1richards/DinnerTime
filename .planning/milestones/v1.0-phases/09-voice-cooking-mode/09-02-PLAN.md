    timerParser.parseTimerPhrase(t: string): number | null
    - "set a timer for 10 minutes" -> 600000
    - "timer 5 min" -> 300000
    - "timer for two minutes" -> 120000
    - "set timer for half an hour" -> 1800000
    - "set a timer for 2 and a half minutes" -> 150000
    - "remind me in 30 seconds" -> 30000

    - "hello world" -> null

    intentRouter.routeIntent(t: string): CookingIntent
    - "next" / "next step" / "continue" / "move on" -> { type: 'next' }
    - "back" / "go back" / "previous" / "last step" -> { type: 'back' }
    - "repeat" / "again" / "say that again" / "what's the step" -> { type: 'repeat' }
    - "pause" / "stop" / "wait" -> { type: 'pause' }
    - "resume" -> { type: 'resume' }
    - "set a timer for 10 minutes" -> { type: 'timer', ms: 600000 } (timer checked before nav verbs so "continue for 10 minutes" can't collide)
    - "what's a substitute for buttermilk" -> { type: 'ask', question: <original transcript> }
    - Empty / whitespace only -> { type: 'ask', question: '' } (caller decides to ignore)
    - Case-insensitive throughout
  </behavior>
  <action>

    Perf test (intentRouter.perf.test.ts): Run 1000 iterations of routeIntent across a mix of 10 phrases (nav verbs, timer phrases, ask phrases) and assert `total_ms < 200`. Acts as a regression guard against accidental Claude calls being introduced into the router.
