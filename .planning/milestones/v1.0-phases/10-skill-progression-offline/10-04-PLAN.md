```typescript
import AsyncStorage from '@react-native-async-storage/async-storage';
import { persist, createJSONStorage } from 'zustand/middleware';

export const useRecipeStore = create<RecipeState>()(
  persist(
    (set, get) => ({ /* existing state */ }),
    {
      name: 'dinnertime-recipes',
      storage: createJSONStorage(() => AsyncStorage),
      partialize: (state) => ({ recipes: state.recipes, lastFetchedAt: state.lastFetchedAt }),
      version: 1,
    }
  )
);
```

```typescript
NetInfo.addEventListener((state) => {

  });
```

Offline queue contract:
```typescript
type QueuedOp =
  | { type: 'markCooked'; entryId: string; recipeId: string }
  | { type: 'pantryEdit'; itemId: string; patch: Partial<PantryItem> };

export const offlineQueue = {
  enqueue(op: QueuedOp): Promise<void>;
  flush(): Promise<{ flushed: number; failed: number }>;
  getPending(): QueuedOp[];
};
```
Queue persists via AsyncStorage under key `dinnertime-offline-queue`.
