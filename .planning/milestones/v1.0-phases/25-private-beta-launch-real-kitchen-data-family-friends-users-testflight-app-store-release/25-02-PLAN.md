   RUN corepack enable

   COPY packages/server/package.json packages/server/
   RUN pnpm install --frozen-lockfile --filter @dinnertime/server...

   COPY . .
   COPY --from=deps /app/node_modules ./node_modules

   ENV NODE_ENV=production
