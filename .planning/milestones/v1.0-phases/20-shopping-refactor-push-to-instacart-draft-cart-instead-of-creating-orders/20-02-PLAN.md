Create a single self-contained component. No tests required — this is pure glue with a UI gesture that's easier to verify manually than via RNTL (the gesture timing is a functional concern covered in DEVICE-TEST-20 ROLLBACK-01).

```typescript
import React, { useCallback, useRef, useState } from 'react';
import { View, Text, Switch, Pressable, StyleSheet } from 'react-native';
import { useSettingsStore } from '../../stores/settingsStore';
import { colors } from '../../design/tokens';

const REVEAL_TAP_COUNT = 5;
const REVEAL_WINDOW_MS = 1500;

export function ShoppingHandoffSection() {
  const mode = useSettingsStore((s) => s.shoppingHandoffMode);
  const setMode = useSettingsStore((s) => s.setShoppingHandoffMode);
  const [revealed, setRevealed] = useState(false);
  const tapLog = useRef<number[]>([]);

  const handleHeaderTap = useCallback(() => {
    const now = Date.now();
    // Keep only taps within the REVEAL_WINDOW
    tapLog.current = [...tapLog.current, now].filter(
      (t) => now - t <= REVEAL_WINDOW_MS,
    );
    if (tapLog.current.length >= REVEAL_TAP_COUNT) {
      setRevealed(true);
      tapLog.current = [];
    }
  }, []);

  return (
    <View className="mt-8">
      <Pressable
        onPress={handleHeaderTap}
        accessibilityLabel="Shopping settings"
        hitSlop={8}
      >
        <Text className="text-xs font-semibold uppercase tracking-wide text-warmGray-500 mb-2">
          Shopping
        </Text>
      </Pressable>

      {!revealed ? (
        <Text className="text-sm text-warmGray-500">
          Items send to Instacart as a draft cart.
        </Text>
      ) : (
        <View className="bg-white rounded-xl border border-warmGray-100 px-4 py-3 flex-row items-center justify-between">
          <View className="flex-1 pr-3">
            <Text className="text-base font-medium text-warmGray-900">
              Use legacy order flow
            </Text>
            <Text className="text-xs text-warmGray-500 mt-0.5">
              Rollback to Phase 8 inline WebBrowser flow. Only flip this if the
              draft-cart handoff is broken in production.
            </Text>
          </View>
          <Switch
            value={mode === 'legacy'}
            onValueChange={(next) => setMode(next ? 'legacy' : 'draft_cart')}
            trackColor={{ false: '#D9D2C7', true: colors.brand }}
            accessibilityLabel="Use legacy order flow"
            accessibilityState={{ checked: mode === 'legacy' }}
          />
        </View>
      )}
    </View>
  );
}
```

Design rules enforced:
- Phase 19 tokens only (NativeWind semantic classes + `colors.brand` from `design/tokens`). No raw hex except the grey track `'#D9D2C7'` which matches the tailwind-config warmGray-200 export — acceptable because Switch's trackColor prop cannot accept className. If the linter complains, import `colors.warmGray200` if defined, else keep the literal with a `// phase-19-exception: Switch prop requires literal` comment.
- Hidden-reveal: 5 taps within 1.5s. Uses a sliding-window array to handle imprecise taps.
- Accessibility: Pressable has accessibilityLabel; Switch has accessibilityState.
- Persist is automatic — `useSettingsStore` writes through to AsyncStorage on state change (configured in 20-00).
