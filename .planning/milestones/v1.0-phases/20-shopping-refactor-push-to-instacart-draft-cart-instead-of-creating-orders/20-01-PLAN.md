import { Linking } from 'react-native';
import * as WebBrowser from 'expo-web-browser';
import { logShoppingEvent, sanitizePayload } from './telemetry';

/**
 * Open an Instacart handoff URL. Prefer native iOS universal-link routing
 * (Linking.openURL) — iOS auto-routes an HTTPS Instacart link to the
 * Instacart app when installed, else Safari. Fall back to Safari View
 * Controller (WebBrowser) when Linking throws (simulators without the
 * real app installed in some configurations).
 *
 * Per SHOP-DC-03 (20-RESEARCH.md Pattern 1). Do NOT use canOpenURL —
 * see 20-RESEARCH.md Pitfall 2 (LSApplicationQueriesSchemes probe
 * requires EAS rebuild).
 */
export async function openInstacartCart(
  url: string,
  opts: { sessionId?: string; orderId?: string | null } = {}
): Promise<void> {
  try {
    await Linking.openURL(url);
    logShoppingEvent({
      name: 'shopping.handoff_opened_app',
      session_id: opts.sessionId ?? '',
      shopping_order_id: opts.orderId ?? null,
      payload: sanitizePayload({ order_id: opts.orderId ?? undefined }),
    });
  } catch {
    await WebBrowser.openBrowserAsync(url);
    logShoppingEvent({
      name: 'shopping.handoff_opened_web',
      session_id: opts.sessionId ?? '',
      shopping_order_id: opts.orderId ?? null,
      payload: sanitizePayload({ order_id: opts.orderId ?? undefined }),
    });
  }
}
```

export type HandoffErrorVariant = 'network' | 'instacart_api' | 'auth';

/**
 * Classify an unknown error into one of three handoff variants. Drives
 * HandoffSheet error-state copy + retry affordance (20-03).
 *
 * Priority order: auth (401/403) → instacart_api (5xx) → network (default).
 * Network catches the "common case" so a flaky fetch reads as retriable.
 */
export function classifyHandoffError(err: unknown): HandoffErrorVariant {
  // Status-bearing object (fetch Response or structured api error)
  if (err && typeof err === 'object' && 'status' in err) {
    const status = Number((err as { status: unknown }).status);
    if (status === 401 || status === 403) return 'auth';
    if (status >= 500 && status < 600) return 'instacart_api';
  }
  // Network-style throwable
  if (err instanceof TypeError) return 'network';
  if (err instanceof Error && /network|fetch/i.test(err.message)) {
    return 'network';
  }
  // Fail-safe: show retry CTA
  return 'network';
}
```
