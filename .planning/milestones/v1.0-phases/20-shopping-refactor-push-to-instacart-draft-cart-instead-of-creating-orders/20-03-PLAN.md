This is a modal bottom-sheet that replaces the Phase 8 inline `ActivityIndicator` + `WebBrowser.openBrowserAsync` flow with a three-state Apple-Pay-style handoff: the app is the curator, Instacart is the checkout. Addresses SHOP-DC-01, SHOP-DC-02, SHOP-DC-06.

```typescript
import React from 'react';
import { Modal, View, Text, Pressable, ActivityIndicator, StyleSheet } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { Button } from '../ui/Button';
import { SymbolIcon } from '../ui/SymbolIcon';
import { colors } from '../../design/tokens';
import type { HandoffErrorVariant } from '../../shopping/classifyHandoffError';

export type HandoffState =
  | { kind: 'idle' }
  | { kind: 'sending' }
  | { kind: 'success'; url: string; itemCount: number; appInstalled: boolean }
  | { kind: 'error'; variant: HandoffErrorVariant; url?: string };

export interface HandoffSheetProps {
  state: HandoffState;
  onOpenCart: () => void;
  onRetry: () => void;
  onDismiss: () => void;
}

const ERROR_COPY: Record<HandoffErrorVariant, { title: string; subtitle: string }> = {
  network: {
    title: "Can't reach Instacart",
    subtitle: 'Check your connection and try again.',
  },
  instacart_api: {
    title: 'Instacart is having trouble',
    subtitle: 'Their servers are returning errors. Try again in a minute.',
  },
  auth: {
    title: 'You need to sign back in',
    subtitle: 'Your Instacart link expired. Try again to refresh it.',
  },
};

export function HandoffSheet({ state, onOpenCart, onRetry, onDismiss }: HandoffSheetProps) {
  const visible = state.kind !== 'idle';
  const dismissable = state.kind !== 'sending';

  return (
    <Modal
      visible={visible}
      transparent
      animationType="slide"
      onRequestClose={dismissable ? onDismiss : undefined}
    >
      <Pressable
        onPress={dismissable ? onDismiss : undefined}
        style={styles.backdrop}
        accessibilityLabel={`Instacart handoff ${state.kind}`}
      >
        {/* Inner Pressable eats taps so the sheet body doesn't dismiss */}
        <Pressable onPress={() => {}}>
          <SafeAreaView edges={['bottom']} style={styles.sheet}>
            {state.kind === 'sending' && (
              <View style={styles.body}>
                <ActivityIndicator size="large" color={colors.brand} />
                <Text style={styles.title}>Sending to Instacart cart…</Text>
                <Text style={styles.subtitle}>Pushing items to your draft cart</Text>
              </View>
            )}

            {state.kind === 'success' && (
              <View style={styles.body}>
                {/* Close button top-right */}
                <Pressable
                  onPress={onDismiss}
                  style={styles.closeBtn}
                  hitSlop={12}
                  accessibilityLabel="Dismiss"
                >
                  <SymbolIcon name="xmark" size={18} tintColor="#8A8078" />
                </Pressable>
                <SymbolIcon
                  name="checkmark.circle.fill"
                  size={56}
                  tintColor={colors.brand}
                />
                <Text style={styles.title}>{state.itemCount} items ready</Text>
                <Text style={styles.subtitle}>
                  Continue in Instacart to pick a delivery window and check out.
                </Text>
                <View style={styles.actions}>
                  <Button
                    title="Open in Instacart"
                    onPress={onOpenCart}
                    variant="primary"
                  />
                  <View style={{ height: 8 }} />
                  <Button
                    title="View shopping list"
                    onPress={onDismiss}
                    variant="ghost"
                  />
                </View>
              </View>
            )}

            {state.kind === 'error' && (
              <View style={styles.body}>
                <Pressable
                  onPress={onDismiss}
                  style={styles.closeBtn}
                  hitSlop={12}
                  accessibilityLabel="Dismiss"
                >
                  <SymbolIcon name="xmark" size={18} tintColor="#8A8078" />
                </Pressable>
                <SymbolIcon
                  name="exclamationmark.triangle"
                  size={48}
                  tintColor={colors.brand}
                />
                <Text style={styles.title}>{ERROR_COPY[state.variant].title}</Text>
                <Text style={styles.subtitle}>
                  {ERROR_COPY[state.variant].subtitle}
                </Text>
                <View style={styles.actions}>
                  <Button title="Try again" onPress={onRetry} variant="primary" />
                  <View style={{ height: 8 }} />
                  <Button title="Cancel" onPress={onDismiss} variant="ghost" />
                </View>
              </View>
            )}
          </SafeAreaView>
        </Pressable>
      </Pressable>
    </Modal>
  );
}

const styles = StyleSheet.create({
  backdrop: {
    flex: 1,
    backgroundColor: 'rgba(0,0,0,0.3)',
    justifyContent: 'flex-end',
  },
  sheet: {
    backgroundColor: '#FAF7F2', // warmWhite — Switch/Modal prop forces literal (Phase 19 exception)
    borderTopLeftRadius: 20,
    borderTopRightRadius: 20,
    paddingHorizontal: 20,
    paddingTop: 32,
    paddingBottom: 16,
    minHeight: 280,
    alignItems: 'center',
  },
  body: {
    width: '100%',
    alignItems: 'center',
  },
  closeBtn: {
    position: 'absolute',
    top: -12,
    right: 0,
    padding: 8,
  },
  title: {
    marginTop: 20,
    fontSize: 22,
    fontWeight: '700',
    color: '#2C231B',
    textAlign: 'center',
  },
  subtitle: {
    marginTop: 8,
    fontSize: 14,
    color: '#5B504A',
    textAlign: 'center',
    paddingHorizontal: 12,
  },
  actions: {
    marginTop: 24,
    width: '100%',
  },
});
```
