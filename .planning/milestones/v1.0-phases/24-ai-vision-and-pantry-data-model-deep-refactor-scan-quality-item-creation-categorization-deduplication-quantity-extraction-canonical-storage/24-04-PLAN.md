function clamp01(n: unknown): number {
  const x = typeof n === 'number' && Number.isFinite(n) ? n : 0.5;
  return Math.max(0, Math.min(1, x));
}

function normalizeFieldConfidence(raw: unknown, flat?: number): FieldConfidence {
  if (raw && typeof raw === 'object') {
    const r = raw as Partial<FieldConfidence>;
    return {
      name:     clamp01(r.name ?? flat ?? 0.5),
      quantity: clamp01(r.quantity ?? flat ?? 0.5),
      unit:     clamp01(r.unit ?? flat ?? 0.5),
      category: clamp01(r.category ?? flat ?? 0.5),
    };
  }
  const f = typeof flat === 'number' ? clamp01(flat) : 0.5;
  return { name: f, quantity: f, unit: f, category: f };
}

function normalizeQuantity(raw: unknown, flatUnit?: string): Quantity {
  // New nested shape
  if (raw && typeof raw === 'object' && 'value' in (raw as any)) {
    return sanitizeQuantity(raw);
  }
  // Legacy flat: quantity was a plain number; unit was a sibling string
  if (typeof raw === 'number') {
    return sanitizeQuantity({ value: raw, unit: flatUnit ?? 'piece', system: 'count' });
  }
  return sanitizeQuantity(null);
}
