```typescript
// getApiBaseUrl() and getAuthToken() are file-local helpers — reuse verbatim.
// startBatchScan is the canonical pattern:
startBatchScan: async (base64Images: string[], sourceLocation: SourceLocation) => {
  set({ isScanning: true });
  try {
    const token = await getAuthToken();
    const response = await fetch(`${getApiBaseUrl()}/api/v1/pantry/scan-batch`, { /* POST with bearer */ });
    if (!response.ok) throw new Error(...);
    const data = await response.json();
    const reviewItems: ReviewItem[] = (data.data ?? []).map((item, index) => ({
      id: `scan-${Date.now()}-${index}`,
      ...item,
      accepted: item.confidence >= 0.7, // same threshold Phase 14
      userEdited: false,
    }));
    set({ scanResults: reviewItems, isScanning: false });
  } catch (err) { set({ isScanning: false }); throw err; }
}
```
