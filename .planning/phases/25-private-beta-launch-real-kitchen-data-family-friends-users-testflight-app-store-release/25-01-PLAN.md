**routes/feedback.ts** — single router file, mirrors telemetry.ts shape:
```ts
import { Hono } from 'hono';
import { z } from 'zod';
import { authMiddleware } from '../middleware/auth.js';
import { env } from '../config/env.js';

const feedback = new Hono();
feedback.use('*', authMiddleware);

const FeedbackSchema = z.object({
  message: z.string().min(1).max(4000),
  email: z.string().email().optional(),
  app_version: z.string().max(32).optional(),
  build_number: z.string().max(32).optional(),
  screenshot_path: z.string().max(512).optional(),
});

feedback.post('/feedback', async (c) => {
  const user = c.get('user') as { id: string, email?: string } | undefined;
  const supabase = c.get('supabase') as any;
  if (!user || !supabase) return c.json({ error: 'unauthorized' }, 401);

  let body: unknown;
  try { body = await c.req.json(); }
  catch { return c.json({ error: 'schema_error', details: 'invalid JSON' }, 400); }

  const parsed = FeedbackSchema.safeParse(body);
  if (!parsed.success) {
    return c.json({ error: 'schema_error', details: parsed.error.flatten() }, 400);
  }

  const row = { ...parsed.data, profile_id: user.id, platform: 'ios' };
  const { data, error } = await supabase.from('feedback_submissions').insert(row).select('id').single();
  if (error) return c.json({ error: 'insert_failed' }, 500);
  return c.json({ id: data.id }, 201);
});

feedback.get('/admin/beta-invites', async (c) => {
  const user = c.get('user') as { id: string, email?: string } | undefined;
  const supabase = c.get('supabase') as any;
  if (!user || !supabase) return c.json({ error: 'unauthorized' }, 401);

  const email = (user.email ?? '').toLowerCase();
  if (!env.ADMIN_EMAILS_LIST.includes(email)) {
    return c.json({ error: 'forbidden' }, 403);
  }

  const { data, error } = await supabase
    .from('beta_invites')
    .select('*')
    .order('invited_at', { ascending: false })
    .limit(100);
  if (error) return c.json({ error: 'query_failed' }, 500);
  return c.json({ invites: data ?? [] }, 200);
});

export default feedback;
```

**FeedbackSheet.tsx** — new file, ~150 lines. Imports:
```tsx
import React, { useState } from 'react';
import { View, Text, TextInput, Modal, Pressable, ActivityIndicator } from 'react-native';
import Constants from 'expo-constants';
import { Button } from '../ui/Button';
import { NetworkErrorBanner } from '../ui/NetworkErrorBanner';
import { authedFetch } from '../../auth/authedFetch';
import { colors } from '../../design/tokens';

export interface FeedbackSheetProps {
  open: boolean;
  onClose: () => void;
  onSuccess?: () => void;  // optional, for test hook + toast
}

export function FeedbackSheet({ open, onClose, onSuccess }: FeedbackSheetProps) {
  const [message, setMessage] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const trimmed = message.trim();
  const canSend = trimmed.length >= 1 && trimmed.length <= 4000 && !submitting;

  const handleSend = async () => {
    setError(null);
    setSubmitting(true);
    try {
      const res = await authedFetch('/api/v1/feedback', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          message: trimmed,
          app_version: Constants.expoConfig?.version,
          build_number: Constants.expoConfig?.ios?.buildNumber,
        }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.error ?? `HTTP ${res.status}`);
      }
      setMessage('');
      onSuccess?.();
      onClose();
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Failed to send');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Modal visible={open} animationType="slide" presentationStyle="pageSheet" onRequestClose={onClose}>
      <View className="flex-1 bg-warmGray-50 pt-4 px-4">
        <View className="flex-row justify-between items-center mb-4">
          <Text className="text-2xl font-bold text-warmGray-900">Send feedback</Text>
          <Pressable onPress={onClose} accessibilityLabel="Cancel">
            <Text className="text-base text-warmGray-600">Cancel</Text>
          </Pressable>
        </View>
        <Text className="text-sm text-warmGray-600 mb-2">
          Tell us what's working, what's broken, or what you wish DinnerTime did.
        </Text>
        <TextInput
          value={message}
          onChangeText={setMessage}
          multiline
          maxLength={4000}
          placeholder="Your feedback…"
          accessibilityLabel="Feedback message"
          className="bg-white border border-warmGray-200 rounded-lg p-3 h-64 text-base text-warmGray-900"
          style={{ textAlignVertical: 'top' }}
        />
        <Text className="text-xs text-warmGray-500 mt-2">{message.length} / 4000</Text>
        {error && <NetworkErrorBanner message={error} onRetry={handleSend} />}
        <View className="mt-4">
          <Button variant="primary" onPress={handleSend} disabled={!canSend}>
            {submitting ? 'Sending…' : 'Send'}
          </Button>
        </View>
      </View>
    </Modal>
  );
}
```
