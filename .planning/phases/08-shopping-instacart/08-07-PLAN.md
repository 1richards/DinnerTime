    ```typescript
    const grouped = useMemo(() => {
      const groups: Partial<Record<GroceryCategory, ShoppingListItem[]>> = {};
      for (const item of items) (groups[item.category] ??= []).push(item);
      return groups;
    }, [items]);
    ```

  - Floating + button opens AddItemSheet

    const { url } = await createOrder();
