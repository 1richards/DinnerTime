```typescript
function makeMockSupabase(opts: {
  canonicals?: Array<{ id: string; canonical_name: string; status?: string }>;
  aliases?: Array<{ canonical_ingredient_id: string; alias_name: string; confidence: number }>;
  onInsertCandidate?: (row: { canonical_name: string }) => { id: string };
}) {
  const canonicalsByName = new Map((opts.canonicals ?? []).map(r => [r.canonical_name, r]));
  const aliasesByName = new Map<string, { canonical_ingredient_id: string; confidence: number }>(
    (opts.aliases ?? []).map(r => [r.alias_name, { canonical_ingredient_id: r.canonical_ingredient_id, confidence: r.confidence }]),
  );
  const calls = { selectCanonicalAll: 0, selectAlias: 0, insert: 0 };

  const from = vi.fn((table: string) => {
    if (table === 'canonical_ingredients') {
      return {
        select: vi.fn((_cols: string) => {
          calls.selectCanonicalAll++;
          const chain: any = {
            then: (resolve: any) => resolve({ data: [...canonicalsByName.values()].filter(r => (r.status ?? 'active') === 'active' || r.status === 'candidate'), error: null }),
          };
          return chain;
        }),
        insert: vi.fn((row: { canonical_name: string }) => {
          calls.insert++;
          const newId = opts.onInsertCandidate?.(row).id ?? 'new-candidate-uuid';
          const inserted = { id: newId, canonical_name: row.canonical_name, status: 'candidate' };
          canonicalsByName.set(row.canonical_name, inserted);
          return { select: vi.fn(() => ({ single: vi.fn(async () => ({ data: inserted, error: null })) })) };
```

Key implementation notes:
- **Cache structure:** `let cache: { names: Array<{id: string; canonical_name: string; status: string}>; fetchedAt: number } | null = null;`. TTL = 60_000ms.
- **Normalize:** `name.trim().toLowerCase()`.
- **Levenshtein:** classic DP, skip if `Math.abs(a.length - b.length) > maxDistance` for early exit. Implement inline; ~30 lines.
- **Candidate INSERT:** `.from('canonical_ingredients').insert({ canonical_name: norm, category: 'other', default_source_location: 'pantry', status: 'candidate' }).select('id').single()`. After success, append to cache (do NOT re-fetch).
- **Batch flow:** dedup rawNames (Set), pre-fetch/refresh cache once, iterate: check exact canonical (cache), check alias (one SELECT per miss), fuzzy (cache scan), else insert candidate. Return Map<input, CanonicalMatch>.
- **Export `_clearCache()`** for test isolation.
- **Confidence constants:** exact canonical = 1.0; exact alias = alias.confidence (passthrough); fuzzy = 0.6; candidate = 0.3.

```typescript
import type { SupabaseClient } from '@supabase/supabase-js';

export type MatchType = 'exact_canonical' | 'exact_alias' | 'fuzzy' | 'candidate_created';
export interface CanonicalMatch { canonicalId: string; matchType: MatchType; confidence: number }

const CACHE_TTL_MS = 60_000;
const FUZZY_MAX_DISTANCE = 2;
const FUZZY_MIN_LEN = 4;

interface CacheRow { id: string; canonical_name: string; status: string }
let cache: { rows: CacheRow[]; fetchedAt: number } | null = null;

export function _clearCache(): void { cache = null; }

async function getCanonicalNames(supabase: SupabaseClient): Promise<CacheRow[]> {
  if (cache && Date.now() - cache.fetchedAt < CACHE_TTL_MS) return cache.rows;
  const { data, error } = await supabase
    .from('canonical_ingredients')
    .select('id, canonical_name, status');
  if (error) throw error;
  cache = { rows: (data ?? []).filter(r => r.status === 'active' || r.status === 'candidate'), fetchedAt: Date.now() };
  return cache.rows;
}

function levenshtein(a: string, b: string, maxDistance: number): number {
  if (Math.abs(a.length - b.length) > maxDistance) return maxDistance + 1;
  // ... standard DP implementation ...
}

export async function resolveCanonical(supabase: SupabaseClient, rawName: string): Promise<CanonicalMatch> {
  const norm = rawName.trim().toLowerCase();
  const rows = await getCanonicalNames(supabase);

  // 1) Exact canonical
  const exact = rows.find(r => r.canonical_name === norm);
  if (exact) return { canonicalId: exact.id, matchType: 'exact_canonical', confidence: 1.0 };

  // 2) Exact alias
  const { data: aliasRows } = await supabase
    .from('ingredient_aliases')
    .select('canonical_ingredient_id, confidence')
    .eq('alias_name', norm)
    .order('confidence', { ascending: false })
    .limit(1);
  if (aliasRows && aliasRows.length > 0) {
    const a = aliasRows[0];
    return { canonicalId: a.canonical_ingredient_id, matchType: 'exact_alias', confidence: a.confidence };
  }

  // 3) Fuzzy — only if name long enough
  if (norm.length >= FUZZY_MIN_LEN) {
    let best: { id: string; distance: number } | null = null;
    for (const r of rows) {
      const d = levenshtein(norm, r.canonical_name, FUZZY_MAX_DISTANCE);
      if (d <= FUZZY_MAX_DISTANCE && (!best || d < best.distance)) best = { id: r.id, distance: d };
    }
    if (best) return { canonicalId: best.id, matchType: 'fuzzy', confidence: 0.6 };
  }

  // 4) Auto-create candidate
  const { data: inserted, error } = await supabase
    .from('canonical_ingredients')
    .insert({ canonical_name: norm, category: 'other', default_source_location: 'pantry', status: 'candidate' })
    .select('id')
    .single();
  if (error || !inserted) throw error ?? new Error('Candidate insert failed');
  if (cache) cache.rows.push({ id: inserted.id, canonical_name: norm, status: 'candidate' });
  return { canonicalId: inserted.id, matchType: 'candidate_created', confidence: 0.3 };
}

export async function resolveCanonicalBatch(
  supabase: SupabaseClient,
  rawNames: string[],
): Promise<Map<string, CanonicalMatch>> {
  const out = new Map<string, CanonicalMatch>();
  const unique = [...new Set(rawNames.map(n => n))]; // preserves raw input keys
  // Pre-warm cache so per-item resolveCanonical reuses it
  await getCanonicalNames(supabase);
  for (const name of unique) {
    out.set(name, await resolveCanonical(supabase, name));
  }
  return out;
}
```
