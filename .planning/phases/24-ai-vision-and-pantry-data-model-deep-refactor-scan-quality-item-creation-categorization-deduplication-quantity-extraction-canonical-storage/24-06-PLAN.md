2. **Implement `resolveFieldClass` in ReviewItemRow.tsx:**
```typescript
import type { FieldConfidence } from '../../stores/pantryStore';

const LOW_CONFIDENCE_THRESHOLD = 0.7;

export function resolveFieldClass(
  fieldConfidence: FieldConfidence | undefined,
  field: keyof FieldConfidence,
): string {
  if (!fieldConfidence) return '';
  const c = fieldConfidence[field];
  if (typeof c !== 'number' || c >= LOW_CONFIDENCE_THRESHOLD) return '';
  return 'border-b border-dashed border-amber-400';
}
```
