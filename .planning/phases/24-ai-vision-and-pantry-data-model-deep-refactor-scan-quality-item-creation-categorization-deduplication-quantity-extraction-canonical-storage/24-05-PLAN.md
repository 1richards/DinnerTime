1. **Rewrite reconcileItems in pantry.ts** (current impl ~line 59-152, keys on normalized_name). New algorithm:

```typescript
import { resolveCanonicalBatch } from './canonicalResolver.js';
import { add as addQuantities, type Quantity } from './units.js';
import type { ScanResult } from './vision.js';

export interface ReconcileResult {
  inserted: number;
  updated: number;
  incompatibleUnits: number;  // rows that created a second pantry_items entry due to unit mismatch
}

export async function reconcileItems(
  supabase: SupabaseClient,
  profileId: string,
  items: ScanResult[],
): Promise<ReconcileResult> {
  if (items.length === 0) return { inserted: 0, updated: 0, incompatibleUnits: 0 };

  // 1. Batch-resolve canonicals
  const resolveMap = await resolveCanonicalBatch(supabase, items.map(it => it.name));

  // 2. Fetch user category overrides in one go (small table, keyed by user)
  const { data: overrides } = await supabase
    .from('canonical_category_override')
    .select('canonical_ingredient_id, category')
    .eq('user_id', profileId);
  const overrideMap = new Map((overrides ?? []).map(o => [o.canonical_ingredient_id, o.category]));

  // 3. Fetch canonical category for all resolved IDs (one query)
  const canonicalIds = [...new Set([...resolveMap.values()].map(m => m.canonicalId))];
  const { data: canonicalRows } = await supabase
    .from('canonical_ingredients')
    .select('id, category')
    .in('id', canonicalIds);
  const canonicalCategoryMap = new Map((canonicalRows ?? []).map(r => [r.id, r.category]));

  let inserted = 0;
  let updated = 0;
  let incompatibleUnits = 0;

  for (const item of items) {
    const match = resolveMap.get(item.name)!;
    const canonicalId = match.canonicalId;
    const resolvedCategory = overrideMap.get(canonicalId) ?? canonicalCategoryMap.get(canonicalId) ?? 'other';

    // Match on identity tuple
    const { data: existingRows } = await supabase
      .from('pantry_items')
      .select('*')
      .eq('profile_id', profileId)
      .eq('canonical_ingredient_id', canonicalId)
      .eq('source_location', item.source_location);

    const existing = existingRows && existingRows.length > 0 ? existingRows[0] : null;

    if (existing) {
      const merged = addQuantities(existing.quantity as Quantity, item.quantity);
      if (merged) {
        await supabase
          .from('pantry_items')
          .update({
            quantity: merged,
            last_seen_at: new Date().toISOString(),
            confidence: item.confidence,
            item_attributes: {
              ...(existing.item_attributes ?? {}),
              canonical_ingredient_id: canonicalId,
              source_location: item.source_location,
            },
          })
          .eq('id', existing.id);
        updated++;
      } else {
        // Incompatible units — insert SECOND row, flag with UX hint
        await supabase.from('pantry_items').insert({
          profile_id: profileId,
          canonical_ingredient_id: canonicalId,
          name: item.name,
          normalized_name: item.normalized_name,
          quantity: item.quantity,
          category: resolvedCategory,
          source_location: item.source_location,
          confidence: item.confidence,
          item_attributes: {
            canonical_ingredient_id: canonicalId,
            source_location: item.source_location,
            reconcile_hint: 'incompatible_units',
          },
          first_seen_at: new Date().toISOString(),
          last_seen_at: new Date().toISOString(),
        });
        incompatibleUnits++;
        inserted++;
      }
    } else {
      // New row
      await supabase.from('pantry_items').insert({
        profile_id: profileId,
        canonical_ingredient_id: canonicalId,
        name: item.name,
        normalized_name: item.normalized_name,
        quantity: item.quantity,
        category: resolvedCategory,
        source_location: item.source_location,
        confidence: item.confidence,
        item_attributes: {
          canonical_ingredient_id: canonicalId,
          source_location: item.source_location,
        },
        first_seen_at: new Date().toISOString(),
        last_seen_at: new Date().toISOString(),
      });
      inserted++;
    }
  }

  return { inserted, updated, incompatibleUnits };
}
```
