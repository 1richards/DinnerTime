    ```typescript
    // apps/mobile/src/app/settings/pantryRulesHelpers.ts
    export interface LocationRule {
      id: string;
      canonical_ingredient_id: string;
      source_location: 'fridge' | 'pantry' | 'freezer';
      precedence: number;
      canonical_name: string;
    }

    export function reorderByIds(
      rules: LocationRule[],
      newIdOrder: string[],
    ): LocationRule[] {
      const byId = new Map(rules.map((r) => [r.id, r]));
      return newIdOrder
        .map((id, idx) => {
          const r = byId.get(id);
          if (!r) return null;
          return { ...r, precedence: idx };
        })
        .filter((x): x is LocationRule => x !== null);
    }
    ```

    ```tsx
    import { SafeAreaView, View, Text, Pressable } from 'react-native';

                  <Text className="flex-1 text-body">{r.alias_name} → {r.canonical_ingredients.canonical_name}</Text>
                  <Pressable onPress={() => deleteRule(r.id)}>
                    <SymbolIcon name="trash" className="text-destructive" />
                  </Pressable>
                </View>
    ```
