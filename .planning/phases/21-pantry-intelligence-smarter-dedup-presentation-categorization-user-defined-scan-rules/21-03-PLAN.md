    ```typescript
    app.post('/suggestions/:id/accept', async (c) => {
      const id = c.req.param('id');
      const { data: suggestion } = await supabase
        .from('suggested_rules')
        .select('rule_type, payload')
        .eq('id', id).eq('user_id', userId).maybeSingle();
      if (!suggestion) return c.json({ error: 'NOT_FOUND' }, 404);
      if (suggestion.rule_type === 'name_mapping') {
        await supabase.from('ingredient_aliases').insert({
          alias_name: suggestion.payload.alias_name,
          canonical_ingredient_id: suggestion.payload.target_canonical_id,
          source: 'user_rule',
          confidence: 1.0,
        });
      } else {
        // location_mapping: need to resolve name→canonical first, or trust
        // that payload already includes canonical_ingredient_id. Use whichever
        // shape suggestionAggregator persists (currently {item_name, user_location}).
        // If item_name is stored rather than canonical_ingredient_id, call
        // canonicalResolver on accept to resolve.
        const { resolveCanonicalBatch } = await import('../services/canonicalResolver.js');
        const [match] = await resolveCanonicalBatch(supabase, [suggestion.payload.item_name]);
        if (match?.canonicalId) {
          await supabase.from('user_location_rules').insert({
            user_id: userId,
            canonical_ingredient_id: match.canonicalId,
            source_location: suggestion.payload.user_location,
            precedence: 0,  // or max+1; pick one, document
          });
        }
      }
      await supabase.from('suggested_rules').update({ dismissed_at: new Date().toISOString() }).eq('id', id);
      return c.json({ ok: true });
    });
    ```

    ```typescript
    app.get('/preview', async (c) => {
      const canonicalId = c.req.query('canonical_id');
      const since = new Date(Date.now() - 30 * 86400_000).toISOString();
      const { data } = await supabase
        .from('scan_events')
        .select('id, final_items, created_at')
        .eq('user_id', userId)
        .gte('created_at', since)
        .order('created_at', { ascending: false })
        .limit(100);
      const { data: target } = await supabase
        .from('canonical_ingredients')
        .select('canonical_name')
        .eq('id', canonicalId).maybeSingle();
      const normalizedTarget = (target?.canonical_name ?? '').toLowerCase().trim();
      const hits: Array<{ name: string; at: string }> = [];
      for (const ev of data ?? []) {
        for (const item of (ev.final_items as any[]) ?? []) {
          if (item.canonical_ingredient_id === canonicalId
              || (item.name ?? '').toLowerCase().trim() === normalizedTarget) {
            hits.push({ name: item.name, at: ev.created_at });
          }
        }
      }
      return c.json({ count: hits.length, items: hits.slice(0, 50) });
    });
    ```
