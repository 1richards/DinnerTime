    ```tsx
    import { ActionSheetIOS, Pressable } from 'react-native';
    import { SymbolIcon } from './SymbolIcon';

    export type EllipsisAction = { label: string; onPress: () => void; destructive?: boolean };

    export function HeaderEllipsis({ actions }: { actions: EllipsisAction[] }) {
      const showSheet = () => {
        const labels = actions.map(a => a.label);
        ActionSheetIOS.showActionSheetWithOptions(
          {
            options: [...labels, 'Cancel'],
            cancelButtonIndex: labels.length,
            destructiveButtonIndex: actions.findIndex(a => a.destructive) >= 0
              ? actions.findIndex(a => a.destructive)
              : undefined,
          },
          (idx) => { if (idx != null && idx < actions.length) actions[idx].onPress(); },
        );
      };
      return (
        <Pressable onPress={showSheet} hitSlop={12} accessibilityLabel="More options">
          <SymbolIcon name="ellipsis" size="body" weight="medium" tintColor="#1F2937" />
        </Pressable>
      );
    }
    ```
