       ```typescript
       import React from 'react';
       import { Pressable, Text, Animated, View, TextInput } from 'react-native';
       import { SymbolView } from 'expo-symbols';
       import { router } from 'expo-router';
       import { colors } from '../../design/tokens';
       import { iconPropsForText } from '../../design/icons';

       export type SearchContext = 'library' | 'something-new' | 'pantry';

       // Pure helpers — testable without RN renderer
       export function buildSearchHref(context: string): string {
         return `/search?context=${context}`;
       }

       export function shadowOpacityConfig(): { inputRange: number[]; outputRange: number[] } {
         return { inputRange: [0, 40], outputRange: [0.05, 0.18] };
       }

       interface StickySearchPillProps {
         placeholder: string;
         context: SearchContext;
         scrollY: Animated.Value;
         onPress?: () =&gt; void;
       }

       export function StickySearchPill({ placeholder, context, scrollY, onPress }: StickySearchPillProps) {
         const cfg = shadowOpacityConfig();
         const shadowOpacity = scrollY.interpolate({ ...cfg, extrapolate: 'clamp' });

         const handlePress = onPress ?? (() =&gt; router.push(buildSearchHref(context)));

         return (
           &lt;Animated.View
             style={{
               position: 'absolute',
               top: 8,
               left: 16,
               right: 16,
               height: 40,
               borderRadius: 20,
               backgroundColor: colors.surface,
               shadowColor: '#000',
               shadowOffset: { width: 0, height: 2 },
               shadowOpacity,
               shadowRadius: 8,
               elevation: 3,
               zIndex: 20,
             }}
           &gt;
             &lt;Pressable
               onPress={handlePress}
               className="flex-1 flex-row items-center px-3"
               accessibilityRole="search"
               accessibilityLabel={placeholder}
             &gt;
               &lt;SymbolView name="magnifyingglass" {...iconPropsForText('caption')} tintColor={colors.textTertiary} /&gt;
               &lt;Text className="ml-2 text-body text-text-tertiary"&gt;{placeholder}&lt;/Text&gt;
             &lt;/Pressable&gt;
           &lt;/Animated.View&gt;
         );
       }

       // Legacy inline SearchBar — used by current recipes/SearchBar call sites until Plan 05 migrates them.
       // Keep the API identical to the old one so no caller breaks.
       interface SearchBarProps {
         value: string;
         onChange: (v: string) =&gt; void;
         placeholder?: string;
       }

       export function SearchBar({ value, onChange, placeholder }: SearchBarProps) {
         return (
           &lt;View className="flex-row items-center bg-surface-subtle rounded-button px-3 h-11"&gt;
             &lt;SymbolView name="magnifyingglass" {...iconPropsForText('caption')} tintColor={colors.textTertiary} /&gt;

               value={value}
               onChangeText={onChange}
               placeholder={placeholder ?? 'Search recipes'}
               placeholderTextColor={colors.textTertiary}
               className="flex-1 ml-2 text-body text-text-primary"
               autoCorrect={false}
               autoCapitalize="none"
               returnKeyType="search"
             /&gt;
             {value.length &gt; 0 ? (
               &lt;Pressable onPress={() =&gt; onChange('')} hitSlop={8}&gt;
                 &lt;SymbolView name="xmark.circle.fill" {...iconPropsForText('caption')} tintColor={colors.textTertiary} /&gt;
               &lt;/Pressable&gt;
             ) : null}
           &lt;/View&gt;
         );
       }
       ```

    ```typescript
    import React from 'react';
    import { Pressable, View, Text } from 'react-native';
    import { SymbolView, type SymbolViewProps } from 'expo-symbols';
    import { Chip, type ChipTone } from './Chip';
    import { colors } from '../../design/tokens';
    import { iconPropsForText } from '../../design/icons';

    type ItemRowLeading =
      | { kind: 'checkbox'; checked: boolean; onToggle: () =&gt; void }
      | { kind: 'stepper'; quantity: number; unit: string | null; onInc: () =&gt; void; onDec: () =&gt; void }
      | { kind: 'icon'; name: SymbolViewProps['name']; tint?: string };

    interface ItemRowProps {
      leading: ItemRowLeading;
      title: string;
      subtitle?: string;
      trailingChip?: { label: string; tone: ChipTone };
      onPress?: () =&gt; void;
      onLongPress?: () =&gt; void;
      struck?: boolean;
    }

    export function ItemRow({ leading, title, subtitle, trailingChip, onPress, onLongPress, struck }: ItemRowProps) {
      const titleCls = `text-body text-text-primary ${struck ? 'line-through opacity-50' : ''}`;

      const Container = onPress || onLongPress ? Pressable : View;

      return (
        &lt;Container
          onPress={onPress}
          onLongPress={onLongPress}
          className="flex-row items-center bg-surface px-4 py-3 border-b border-border-subtle min-h-[56px]"
        &gt;
          {/* Leading */}
          &lt;View className="mr-3"&gt;
            {leading.kind === 'checkbox' ? (
              &lt;Pressable
                onPress={leading.onToggle}
                hitSlop={8}
                className={`w-6 h-6 rounded-button border items-center justify-center ${
                  leading.checked ? 'bg-brand border-brand' : 'bg-surface border-border'
                }`}
                accessibilityRole="checkbox"
                accessibilityState={{ checked: leading.checked }}
              &gt;
                {leading.checked ? (
                  &lt;SymbolView name="checkmark" size={14} weight="bold" tintColor="#FFFFFF" /&gt;
                ) : null}
              &lt;/Pressable&gt;
            ) : leading.kind === 'stepper' ? (
              &lt;View className="flex-row items-center"&gt;
                &lt;Pressable
                  onPress={leading.onDec}
                  hitSlop={8}
                  disabled={leading.quantity &lt;= 0}
                  className="w-8 h-8 rounded-button bg-surface-subtle items-center justify-center"
                  accessibilityLabel="Decrease quantity"
                &gt;
                  &lt;SymbolView name="minus" {...iconPropsForText('caption')} tintColor={colors.textPrimary} /&gt;
                &lt;/Pressable&gt;
                &lt;Text className="mx-2 text-body text-text-primary"&gt;
                  {leading.quantity}{leading.unit ? ` ${leading.unit}` : ''}
                &lt;/Text&gt;
                &lt;Pressable
                  onPress={leading.onInc}
                  hitSlop={8}
                  className="w-8 h-8 rounded-button bg-surface-subtle items-center justify-center"
                  accessibilityLabel="Increase quantity"
                &gt;
                  &lt;SymbolView name="plus" {...iconPropsForText('caption')} tintColor={colors.textPrimary} /&gt;
                &lt;/Pressable&gt;
              &lt;/View&gt;
            ) : (
              &lt;SymbolView
                name={leading.name}
                {...iconPropsForText('body')}
                tintColor={leading.tint ?? colors.textSecondary}
              /&gt;
            )}
          &lt;/View&gt;

          {/* Title + subtitle */}
          &lt;View className="flex-1"&gt;
            &lt;Text className={titleCls} numberOfLines={1}&gt;{title}&lt;/Text&gt;
            {subtitle ? &lt;Text className="text-caption text-text-secondary" numberOfLines={1}&gt;{subtitle}&lt;/Text&gt; : null}
          &lt;/View&gt;

          {/* Trailing chip */}
          {trailingChip ? (
            &lt;Chip label={trailingChip.label} kind="display" tone={trailingChip.tone} /&gt;
          ) : null}
        &lt;/Container&gt;
      );
    }
    ```

    ```typescript
    export interface TitleClassArgs { struck?: boolean }
    export function resolveTitleClasses({ struck }: TitleClassArgs): string {

    export function resolveCheckboxBoxClasses({ checked }: CheckboxBoxArgs): string {
    ```
